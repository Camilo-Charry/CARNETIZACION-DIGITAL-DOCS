\section{Detalles Técnicos de Implementación}
\subsection{Arquitectura del Sistema}
El sistema implementado sigue una arquitectura cliente-servidor con los siguientes componentes:

\begin{itemize}
  \item \textbf{Aplicación Móvil}: Desarrollada en Android Studio con Java y Firebase para almacenamiento en la nube.
  \item \textbf{Lector QR/RFID}: Integración con hardware NFC para lectura de tarjetas RFID.
  \item \textbf{Servidor Backend}: API REST desarrollada en Node.js con base de datos MongoDB.
  \item \textbf{Panel de Administración}: Interfaz web para gestión de usuarios y reportes.
\end{itemize}

\subsection{Algoritmos de Seguridad}
Se implementaron los siguientes mecanismos de seguridad:

\begin{enumerate}
  \item \textbf{Encriptación AES-256}: Para protección de datos sensibles en tránsito y almacenamiento.
  \item \textbf{Códigos QR Dinámicos}: Generación de códigos con validez temporal de 30 segundos.
  \item \textbf{Autenticación Multifactor}: Combinación de credenciales y verificación biométrica opcional.
  \item \textbf{Logs de Auditoría}: Registro detallado de todas las operaciones del sistema.
\end{enumerate}

\section{Código Fuente}
\subsection{Ejemplo de Implementación QR}
\begin{verbatim}
// Generación de código QR dinámico
public String generateDynamicQR(String userId, long timestamp) {
    String payload = userId + ":" + timestamp + ":" + generateNonce();
    String encrypted = AES256.encrypt(payload, getSecretKey());
    return QRCodeGenerator.generate(encrypted, 256, 256);
}

// Validación de código QR
public boolean validateQR(String qrCode, String userId) {
    try {
        String decrypted = AES256.decrypt(qrCode, getSecretKey());
        String[] parts = decrypted.split(":");
        long timestamp = Long.parseLong(parts[1]);
        return parts[0].equals(userId) &&
               (System.currentTimeMillis() - timestamp) < 30000; // 30s
    } catch (Exception e) {
        return false;
    }
}
\end{verbatim}

\subsection{Ejemplo de Implementación RFID}
\begin{verbatim}
// Lectura de tarjeta RFID
public String readRFIDTag(NfcAdapter nfcAdapter) {
    NfcTag tag = nfcAdapter.getTag();
    if (tag != null) {
        Ndef ndef = Ndef.get(tag);
        if (ndef != null) {
            ndef.connect();
            NdefMessage message = ndef.getNdefMessage();
            ndef.close();
            return new String(message.getRecords()[0].getPayload());
        }
    }
    return null;
}

// Verificación de acceso RFID
public boolean verifyRFIDAccess(String tagData, String location) {
    // Verificar validez del tag
    if (!isValidTag(tagData)) return false;

    // Verificar permisos de ubicación
    User user = getUserFromTag(tagData);
    return user.hasAccessTo(location);
}
\end{verbatim}

\section{Limitaciones del Estudio}
\subsection{Limitaciones Técnicas}
\begin{itemize}
  \item \textbf{Cobertura de Red}: El sistema requiere conectividad a internet para sincronización \cite{ref11,ref12}.
  \item \textbf{Compatibilidad de Dispositivos}: Limitado a dispositivos Android con NFC \cite{ref13}.
  \item \textbf{Escalabilidad}: Probado hasta 5000 usuarios concurrentes \cite{ref14}.
  \item \textbf{Costos de Hardware}: Lectores RFID representan inversión inicial significativa \cite{ref15,ref16}.
\end{itemize}

\subsection{Limitaciones Metodológicas}
\begin{itemize}
  \item \textbf{Muestra}: Estudio realizado en una sola institución educativa \cite{ref17}.
  \item \textbf{Tiempo}: Período de evaluación de 6 meses \cite{ref18}.
  \item \textbf{Variables Controladas}: Dificultad para aislar variables en entorno real \cite{ref19}.
  \item \textbf{Sesgo de Selección}: Participantes voluntarios pueden no representar población general \cite{ref20}.
\end{itemize}
